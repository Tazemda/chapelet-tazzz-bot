// ---------- DESSIN CHAPELET AUTOUR DU CADRE ---------- function dessinerChapelet() { const canvas = document.getElementById('chapeletCanvas'); const wrapper = document.querySelector('.chapelet-wrapper'); const content = document.getElementById('jourContent'); if (!canvas || !wrapper || !content) return; const rect = content.getBoundingClientRect(); const wrapperRect = wrapper.getBoundingClientRect(); // taille réelle du cadre const width = rect.width; const height = rect.height; if (width === 0 || height === 0) return; canvas.width = wrapperRect.width; canvas.height = wrapperRect.height; const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); // On va placer les éléments tout autour du rectangle, à une distance fixe des bords // Paramètres : 5 séries de 10 petits ronds + 1 grand rond entre chaque série // Total = 55 ronds const petitRayon = Math.min(width, height) * 0.025; // 2.5% de la plus petite dimension const grandRayon = petitRayon * 1.6; const distanceBord = petitRayon * 1.2; // écart du bord du cadre // Périmètre du rectangle (intérieur, là où on pose les ronds) // On parcourt le tour : haut, droite, bas, gauche const topY = distanceBord; const bottomY = height - distanceBord; const leftX = distanceBord; const rightX = width - distanceBord; // Longueur des côtés const topLen = rightX - leftX; const rightLen = bottomY - topY; const bottomLen = topLen; const leftLen = rightLen; const perimetre = topLen + rightLen + bottomLen + leftLen; // On veut 55 positions régulièrement espacées sur ce périmètre const nbPositions = 55; const step = perimetre / nbPositions; // Tableau des positions [x,y] le long du périmètre let positions = []; let current = 0; // côté haut (de gauche à droite) for (let x = leftX; x <= rightX && current <= perimetre; x += step) { if (x > rightX) x = rightX; positions.push({ x, y: topY }); if (x === rightX) break; current = x - leftX; if (current + step > topLen) break; } current = topLen; // côté droit (de haut en bas) for (let y = topY + step; y <= bottomY && current <= perimetre; y += step) { if (y > bottomY) y = bottomY; positions.push({ x: rightX, y }); if (y === bottomY) break; current += step; if (current + step > topLen + rightLen) break; } current = topLen + rightLen; // côté bas (de droite à gauche) for (let x = rightX - step; x >= leftX && current <= perimetre; x -= step) { if (x < leftX) x = leftX; positions.push({ x, y: bottomY }); if (x === leftX) break; current += step; if (current + step > topLen + rightLen + bottomLen) break; } current = topLen + rightLen + bottomLen; // côté gauche (de bas en haut) for (let y = bottomY - step; y > topY && current <= perimetre; y -= step) { if (y < topY) y = topY; positions.push({ x: leftX, y }); if (y === topY) break; current += step; if (current + step > perimetre) break; } // Tronquer à 55 positions exactement positions = positions.slice(0, 55); // Maintenant, colorier en fonction du rang : 5 séries de 10 petits + 1 grand for (let i = 0; i < positions.length; i++) { const pos = positions[i]; // déterminer si c'est un grand rond (indices 10, 21, 32, 43, 54 ?) // 5 séries de 10 : après 10,20,30,40,50 on ajoute un grand const isGrand = (i+1) % 11 === 0; // 11e, 22e, 33e, 44e, 55e const rayon = isGrand ? grandRayon : petitRayon; ctx.beginPath(); ctx.arc(pos.x, pos.y, rayon, 0, 2 * Math.PI); ctx.fillStyle = isGrand ? "#e67e22" : "#f1c40f"; ctx.fill(); ctx.strokeStyle = "#2c3e4e"; ctx.lineWidth = 1; ctx.stroke(); } } // Observer les changements de taille du cadre function initChapeletObserver() { const content = document.getElementById('jourContent'); if (!content) return; const resizeObserver = new ResizeObserver(() => dessinerChapelet()); resizeObserver.observe(content); window.addEventListener('resize', () => dessinerChapelet()); dessinerChapelet(); // dessin initial } // Appeler après le chargement de la page if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', initChapeletObserver); } else { initChapeletObserver(); }
